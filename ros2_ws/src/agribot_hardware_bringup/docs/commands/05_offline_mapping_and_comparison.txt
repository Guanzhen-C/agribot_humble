# Jetson离线建图和定位轨迹对比

# 首次安装LIO-SAM离线依赖
cd /home/cgz/agribot_ws/ros2_ws
source /opt/ros/humble/setup.bash
vcs import src < src/agribot_offline_mapping/third_party.repos
rosdep install --from-paths src/LIO-SAM src/agribot_offline_mapping \
  --ignore-src -r -y
colcon build --packages-select lio_sam agribot_offline_mapping \
  --symlink-install --parallel-workers "$(nproc)"

# 有RTK数据：强约束LIO-SAM建图、二维投影、地理配准和轨迹保存
source /opt/ros/humble/setup.bash
source /home/cgz/agribot_ws/ros2_ws/install/setup.bash
export BAG_PATH=/media/cgz/data/agribot_data/bags/这里改成实际数据包目录名
export MAP_NAME=map_lio_sam_这里改成新地图名
export MAP_BASE=/media/cgz/data/agribot_data/maps/$MAP_NAME
ros2 run agribot_offline_mapping run_rtk_mapping_pipeline.py \
  "$BAG_PATH" "$MAP_BASE" --domain-id 71 --playback-rate 0.5

# 无RTK数据：LIO-SAM三维图和二维导航图
ros2 run agribot_offline_mapping run_rtk_mapping_pipeline.py \
  "$BAG_PATH" "$MAP_BASE" --playback-rate 0.5 --without-rtk

# 已存在同名结果并确认需要覆盖时，在命令末尾增加--force。
ls -lh ${MAP_BASE}.pcd ${MAP_BASE}.pgm ${MAP_BASE}.yaml \
  ${MAP_BASE}_manifest.yaml
ls -lh ${MAP_BASE}_georeference.yaml 2>/dev/null || true
ros2 bag info ${MAP_BASE}_result

# 重新计算FAST-LIO2、FAST-LIVO2、KF-GINS和robot_localization轨迹
ros2 run agribot_offline_mapping run_localization_comparison.py \
  "$BAG_PATH" "${MAP_BASE}_comparison" \
  --domain-id 74 --playback-rate 0.5

# 显示二维图、三维图和定位轨迹
export DISPLAY=:1
export XAUTHORITY=/run/user/1000/gdm/Xauthority
ros2 launch agribot_offline_mapping lio_sam_rtk_result.launch.py \
  map_base:=$MAP_BASE rviz:=true \
  show_3d_map:=true show_comparison_paths:=true
