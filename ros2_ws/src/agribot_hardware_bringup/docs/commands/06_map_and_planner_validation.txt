# 地图查看、手工清理和Smac规划验证

source /opt/ros/humble/setup.bash
source /home/cgz/agribot_ws/ros2_ws/install/setup.bash
export DISPLAY=:1
export XAUTHORITY=/run/user/1000/gdm/Xauthority
export MAP_BASE=/media/cgz/data/agribot_data/maps/map_lio_sam_0811

# 手工清理二维PGM前先备份；不要缩放、裁剪或改变图像尺寸
cp "${MAP_BASE}.pgm" \
  "${MAP_BASE}_manual_backup_$(date +%Y%m%d_%H%M%S).pgm"
gimp "${MAP_BASE}.pgm"
# GIMP中按P选择画笔，按X切换白色；完成后按Ctrl+E覆盖导出PGM。
file "${MAP_BASE}.pgm"

# Smac阿克曼路径验证；二维地图用于规划，三维图只用于显示
ros2 launch agribot_hardware_bringup \
  ackermann_smac_planner_validation.launch.py \
  map:=${MAP_BASE}.yaml pcd_map:=${MAP_BASE}.pcd \
  show_3d_map:=true rviz:=true

# RViz先点2D Pose Estimate设起点，再连续点2D Goal Pose添加经停点。
# 该入口不启动控制器、传感器或底盘。

# Nav2原生Nav Through Poses连续规划验证
ros2 launch agribot_hardware_bringup \
  ackermann_nav_through_poses_validation.launch.py \
  map:=${MAP_BASE}.yaml pcd_map:=${MAP_BASE}.pcd \
  show_3d_map:=true rviz:=true
