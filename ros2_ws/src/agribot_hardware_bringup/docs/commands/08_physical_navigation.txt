# 通用真机导航入口
# 同一时间只运行一套整车launch；桌面CAN GUI不能与CAN底盘节点同时占用USB-CAN。

source /opt/ros/humble/setup.bash
source /home/sunrise/agribot_ws/ros2_ws/install/setup.bash
export ROS_DOMAIN_ID=0
export ROS_LOCALHOST_ONLY=0
export DISPLAY=:0
export XAUTHORITY=/home/sunrise/.Xauthority
export ZQWL_PORT=/dev/serial/by-id/usb-ZQWL-CANFD_ZQWL-CANFD_966960660237-if00

# FAST-LIO局部滚动导航：不使用保存地图，目标必须位于滚动窗口内
ros2 launch agribot_hardware_bringup ackermann_mppi_fastlio_local.launch.py \
  rviz:=true enable_chassis_output:=false \
  chassis_driver:=ackermann_can can_transport:=zqwl_cdc \
  zqwl_port:=$ZQWL_PORT

# FAST-LIO保存地图导航：RViz手动粗定位后执行NDT/GICP
export MAP_BASE=/home/sunrise/agribot_maps/test_site/这里改成地图名
ros2 launch agribot_hardware_bringup ackermann_mppi_fastlio_mapped.launch.py \
  map_base:=$MAP_BASE initialization_source:=manual rviz:=true \
  enable_chassis_output:=false chassis_driver:=ackermann_can \
  can_transport:=zqwl_cdc zqwl_port:=$ZQWL_PORT

# NavSat/KF-GINS保存地图导航
ros2 launch agribot_hardware_bringup ackermann_mppi_navsat.launch.py \
  map:=${MAP_BASE}.yaml \
  map_georeference:=${MAP_BASE}_georeference.yaml \
  initialization_source:=rtk rviz:=true \
  enable_chassis_output:=false chassis_driver:=ackermann_can \
  can_transport:=zqwl_cdc zqwl_port:=$ZQWL_PORT

# FAST-LIVO2+RTK通用保存地图导航
ros2 launch agribot_hardware_bringup \
  ackermann_mppi_fastlivo_rtk_mapped.launch.py \
  map_base:=$MAP_BASE initialization_source:=manual \
  allow_missing_georeference:=true start_rtk:=true rviz:=true \
  enable_chassis_output:=false chassis_driver:=ackermann_can \
  can_transport:=zqwl_cdc zqwl_port:=$ZQWL_PORT

# 有有效_georeference.yaml并且RTK位置质量4、双天线航向整数固定时，
# 可把initialization_source改为rtk，并把allow_missing_georeference改为false。

# 分布式显示：RDK命令将rviz改为false；Jetson另开终端执行
source /opt/ros/humble/setup.bash
source /home/cgz/agribot_ws/ros2_ws/install/setup.bash
export ROS_DOMAIN_ID=0

ros2 daemon stop || true
ros2 daemon start
ros2 run rviz2 rviz2 \
