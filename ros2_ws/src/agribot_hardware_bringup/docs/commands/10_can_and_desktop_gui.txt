# USB-CAN和桌面控制GUI
# 桌面GUI与ROS底盘节点不能同时运行；二者都会独占USB-CAN串口。

source /opt/ros/humble/setup.bash
source /home/sunrise/agribot_ws/ros2_ws/install/setup.bash
export ZQWL_PORT=/dev/serial/by-id/usb-ZQWL-CANFD_ZQWL-CANFD_966960660237-if00

# 查看稳定设备名
ls -l $ZQWL_PORT

# 只发送零速帧并检查0x101、0x102、0x103反馈，不让车辆运动
ros2 run agribot_hardware_bringup wheeltec_car_gui.py \
  --port $ZQWL_PORT --link-test --link-test-duration 3

# 查找USB-CAN占用者；有输出时先正常关闭对应程序
fuser -v $ZQWL_PORT

# 启动桌面GUI
ros2 run agribot_hardware_bringup start_wheeltec_car_gui.sh

# 当前GUI默认速度0.30m/s，最大0.50m/s；默认和最大转角均为0.30rad。
# 解锁后按住方向按钮或W/A/S/D才持续发送，松开立即发送零速。
