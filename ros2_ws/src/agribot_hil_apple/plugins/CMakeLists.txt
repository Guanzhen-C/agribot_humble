cmake_minimum_required(VERSION 3.16)
project(agribot_hil_sim_plugins)
find_package(ament_cmake REQUIRED)
find_package(gazebo_dev REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
add_library(agribot_hil_gripper_hold SHARED gripper_hold.cpp)
target_compile_features(agribot_hil_gripper_hold PRIVATE cxx_std_17)
ament_target_dependencies(agribot_hil_gripper_hold gazebo_dev gazebo_ros rclcpp std_msgs)
add_library(agribot_hil_work_motors SHARED work_motors.cpp)
target_compile_features(agribot_hil_work_motors PRIVATE cxx_std_17)
ament_target_dependencies(agribot_hil_work_motors gazebo_dev gazebo_ros rclcpp std_msgs sensor_msgs)
add_library(agribot_hil_work_spray_visual SHARED work_spray_visual.cpp)
target_compile_features(agribot_hil_work_spray_visual PRIVATE cxx_std_17)
ament_target_dependencies(agribot_hil_work_spray_visual gazebo_dev)
target_link_libraries(agribot_hil_work_spray_visual ${GAZEBO_LIBRARIES})
add_library(agribot_hil_work_spray_obstacles SHARED work_spray_obstacles.cpp)
target_compile_features(agribot_hil_work_spray_obstacles PRIVATE cxx_std_17)
ament_target_dependencies(agribot_hil_work_spray_obstacles gazebo_dev)
target_link_libraries(agribot_hil_work_spray_obstacles ${GAZEBO_LIBRARIES})
