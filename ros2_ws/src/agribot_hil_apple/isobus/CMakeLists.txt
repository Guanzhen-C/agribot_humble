cmake_minimum_required(VERSION 3.16)
project(agribot_hil_isobus LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(AGISOSTACK_ROOT "$ENV{HOME}/.cache/agribot/isobus/AgIsoStack-plus-plus" CACHE PATH "Pinned stack checkout")
execute_process(COMMAND git -C "${AGISOSTACK_ROOT}" rev-parse HEAD
    OUTPUT_VARIABLE AGISOSTACK_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE GIT_RESULT)
if(NOT GIT_RESULT EQUAL 0 OR NOT AGISOSTACK_COMMIT STREQUAL "02f3fe910de5fdc73e2014394c2e80900490b9cc")
    message(FATAL_ERROR "Expected pinned AgIsoStack++ revision; run scripts/fetch_isobus_stack.sh")
endif()
set(CAN_STACK_DISABLE_THREADS ON CACHE BOOL "Single-threaded MCU-compatible stack" FORCE)
set(DISABLE_CAN_STACK_LOGGER ON CACHE BOOL "Application owns diagnostics" FORCE)
set(DISABLE_ISOBUS_DATA_DICTIONARY ON CACHE BOOL "Private PGN prototype, not TC" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "No upstream test dependencies" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "Only HIL endpoint" FORCE)
add_subdirectory("${AGISOSTACK_ROOT}" agisostack EXCLUDE_FROM_ALL)
add_library(hil_work_protocol STATIC work_protocol.c)
target_include_directories(hil_work_protocol PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_library(hil_isobus_core STATIC endpoint.cpp)
target_link_libraries(hil_isobus_core PUBLIC hil_work_protocol Isobus Utility)
target_include_directories(hil_isobus_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(hil_isobus_endpoint host_main.cpp)
target_link_libraries(hil_isobus_endpoint PRIVATE hil_isobus_core)
target_compile_options(hil_isobus_core PRIVATE -Wall -Wextra -Werror)
target_compile_options(hil_isobus_endpoint PRIVATE -Wall -Wextra -Werror)
