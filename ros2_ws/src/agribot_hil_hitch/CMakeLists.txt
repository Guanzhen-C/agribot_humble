cmake_minimum_required(VERSION 3.16)
project(agribot_hil_hitch LANGUAGES C CXX)
find_package(ament_cmake REQUIRED)
find_package(gazebo_dev REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
add_library(agribot_hil_hitch_plant SHARED plugins/hitch_plant.cpp)
target_compile_features(agribot_hil_hitch_plant PRIVATE cxx_std_17)
ament_target_dependencies(agribot_hil_hitch_plant gazebo_dev gazebo_ros rclcpp std_msgs sensor_msgs)
add_library(agribot_hil_hitch_tillage SHARED plugins/tillage_visual.cpp)
target_compile_features(agribot_hil_hitch_tillage PRIVATE cxx_std_17)
ament_target_dependencies(agribot_hil_hitch_tillage gazebo_dev)
target_link_libraries(agribot_hil_hitch_tillage ${GAZEBO_LIBRARIES})
add_library(agribot_hil_hitch_linkage SHARED plugins/linkage_visual.cpp)
target_compile_features(agribot_hil_hitch_linkage PRIVATE cxx_std_17)
ament_target_dependencies(agribot_hil_hitch_linkage gazebo_dev)
target_link_libraries(agribot_hil_hitch_linkage ${GAZEBO_LIBRARIES})
add_library(agribot_hil_powertrain_plant SHARED plugins/powertrain_plant.cpp protocol/powertrain_plant.c protocol/powertrain.c protocol/hitch_control.c)
target_compile_features(agribot_hil_powertrain_plant PRIVATE cxx_std_17)
target_include_directories(agribot_hil_powertrain_plant PRIVATE protocol)
ament_target_dependencies(agribot_hil_powertrain_plant gazebo_dev gazebo_ros rclcpp std_msgs)
install(TARGETS agribot_hil_hitch_plant agribot_hil_hitch_tillage agribot_hil_hitch_linkage agribot_hil_powertrain_plant LIBRARY DESTINATION lib)
install(DIRECTORY scripts protocol firmware launch DESTINATION share/${PROJECT_NAME}
  PATTERN "__pycache__" EXCLUDE PATTERN "*.pyc" EXCLUDE)
install(FILES README.md POWERTRAIN.md POWERTRAIN_PROTOCOL.md POWERTRAIN_TEST_REPORT.md DESTINATION share/${PROJECT_NAME})
install(DIRECTORY docs DESTINATION share/${PROJECT_NAME})
ament_package()
