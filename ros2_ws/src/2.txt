# 航向解优先连续稳定为 SOL_COMPUTED,L1_INT 或 SOL_COMPUTED,NARROW_INT
# L1_FLOAT也可供ESKF使用，但会自动采用不小于5度的标准差降低权重；正式精度测试应等待整数固定解
ros2 topic echo /rtk/heading_solution --field data

# 航向有效标志应持续为 true；true 本身不代表已经达到整数固定解
ros2 topic echo /rtk/heading_valid --field data

# 主天线在左、副天线在右；车头航向为北0度、顺时针为正
# 车辆静止时航向角不应明显跳变
ros2 topic echo /rtk/heading_deg --field data

# 查看航向及其动态协方差；pose.covariance[35]是航向方差，单位为rad^2
# 整数固定解下限约0.0003046（1度标准差），浮点解下限约0.0076154（5度标准差）
ros2 topic echo /rtk/heading_with_covariance --once
