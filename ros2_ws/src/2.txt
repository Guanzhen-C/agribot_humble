# 终端 3：查看数据；以下命令每次任选一条，按 Ctrl+C 后再查看下一项
source /opt/ros/humble/setup.bash
source /home/sunrise/agribot_ws/ros2_ws/install/setup.bash
export ROS_DOMAIN_ID=0
export ROS_LOCALHOST_ONLY=0

# 查看RTK主天线的原始经度、纬度、高度和协方差
ros2 topic echo /rtk/fix

# 主天线在左、副天线在右；减90度后得到车头航向，北0度、顺时针为正
ros2 topic echo /rtk/heading_deg

# 查看同一车头航向的ROS ENU四元数；这是ESKF实际使用的话题
ros2 topic echo /rtk/heading

# 查看位置解质量；data: 4 表示 RTK 固定解
ros2 topic echo /rtk/fix_quality

# 查看航向解状态；SOL_COMPUTED,NARROW_INT 表示航向整数固定解
ros2 topic echo /rtk/heading_solution

# 查看航向是否有效；data: true 表示有效
ros2 topic echo /rtk/heading_valid

# 查看RTK测量转换到局部ENU后的原始位姿；第一帧固定解是当前临时原点
ros2 topic echo /navsat_pose

# 查看ESKF融合定位；pose是后轴中心位姿，twist是融合速度
ros2 topic echo /odometry/filtered_navsat

# 分别检查RTK、IMU和ESKF输出频率
ros2 topic hz /rtk/fix
ros2 topic hz /imu/data
ros2 topic hz /odometry/filtered_navsat
